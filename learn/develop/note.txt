1 目录准备
sudo mkdir -p /opt/{mysql,redis}/{data,conf,logs}
sudo chmod -R 777 /opt/mysql /opt/redis

2 MySQL 开发配置（/opt/mysql/conf/my-dev.cnf）
[mysqld]
# 基础
server_id        = 1
port             = 3306
datadir          = /var/lib/mysql
socket           = /var/run/mysqld/mysqld.sock
character_set_server = utf8mb4
collation_server     = utf8mb4_unicode_ci
default_time_zone    = +08:00

# 内存——2 GB 机器开发档
innodb_buffer_pool_size      = 512M      # 最大块，占 25 %
innodb_log_file_size         = 128M
innodb_log_buffer_size       = 32M
key_buffer_size              = 32M
tmp_table_size               = 32M
max_heap_table_size          = 32M
sort_buffer_size             = 512K
join_buffer_size             = 512K
read_buffer_size             = 512K
read_rnd_buffer_size         = 512K
thread_cache_size            = 50
max_connections              = 100       # 开发足够

# 功能开关
performance_schema           = OFF       # 省 100+ MB
skip_name_resolve            = ON
lower_case_table_names       = 1

# 日志（开发调试用）
log_error                    = /var/log/mysql/error.log
slow_query_log               = 1
slow_query_log_file          = /var/log/mysql/slow.log
long_query_time              = 1

3 Redis 开发配置（/opt/redis/conf/redis-dev.conf）
# 网络
bind 0.0.0.0
port 6379
protected-mode yes
requirepass Qwer12356

# 内存——2 GB 机器
maxmemory 512mb
maxmemory-policy allkeys-lru

# 持久化（开发可关，省 IO）
save ""
appendonly no

# 日志
logfile /var/log/redis/redis.log
loglevel notice

# 其他
databases 16
tcp-keepalive 300

4 创建docker网络
docker network create sf-net

5 验证
# 内存占用
docker stats mysql-dev redis-dev --no-stream

# 连接测试
mysql -h127.1 -uroot -pQwer12356 -e "SELECT 1;"
redis-cli -a Qwer12356 PING

nginx重载
nginx配置路径：/etc/nginx/conf.d
nginx -t
systemctl reload nginx
