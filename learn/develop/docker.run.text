4.1 Docker 启动
# MySQL
docker network create sf-net
docker run -d --name mysql --network sf-net --restart=unless-stopped \
  -p 3306:3306 \
  -e MYSQL_ROOT_PASSWORD=Qwer12356 \
  -e TZ=Asia/Shanghai \
  -v /opt/mysql/conf/mysql-dev.cnf:/etc/mysql/conf.d/z-dev.cnf:ro \
  -v /opt/mysql/data:/var/lib/mysql \
  -v /opt/mysql/logs:/var/log/mysql \
  mysql:8.0

# Redis
docker run -d --name redis --network sf-net --restart=unless-stopped \
  -p 6379:6379 \
  -v /opt/redis/conf/redis-dev.conf:/usr/local/etc/redis/redis.conf:ro \
  -v /opt/redis/data:/data \
  -v /opt/redis/logs:/var/log/redis \
  redis:7-alpine redis-server /usr/local/etc/redis/redis.conf

# minIO
docker run -d --name minio --network sf-net --restart=unless-stopped \
  -p 9000:9000 -p 9001:9001 \
  -v /opt/minio/data:/data \
  -e "MINIO_ROOT_USER=minio" \
  -e "MINIO_ROOT_PASSWORD=Qwer12356" \
  minio/minio:latest server /data --console-address ":9001"

//打包镜像

//启动镜像
docker run -d --name tongcheng-web --network sf-net --restart=unless-stopped \
    -p 12001:12001 \
    -v /opt/tongcheng/logs/web:/app/service/web/logs \
    -v /opt/tongcheng/conf/web:/app/service/web/conf \
  tongcheng-web:latest

docker run -d --name tongcheng-admin --network sf-net --restart=unless-stopped \
    -p 12002:12002 \
    -v /opt/tongcheng/logs/admin:/app/service/admin/logs \
    -v /opt/tongcheng/conf/admin:/app/service/admin/conf \
  tongcheng-admin:latest
